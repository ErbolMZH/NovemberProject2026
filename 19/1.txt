CREATE OR REPLACE FUNCTION log_subject_diff()
RETURNS TRIGGER AS $$
DECLARE
    old_json JSONB;
    new_json JSONB;
    v_row    JSONB;
    diff_old JSONB := '{}'::JSONB;
    diff_new JSONB := '{}'::JSONB;
    k TEXT;
    v_old JSONB;
    v_new JSONB;
    val_text TEXT;

    v_step        TEXT := 'start';
    v_sid         UUID;
    v_record_id   TEXT;
    v_err_state   TEXT;
    v_err_msg     TEXT;
    v_err_detail  TEXT;
    v_err_context TEXT;
BEGIN
    v_step := 'определение subject_id';
    v_row := CASE WHEN TG_OP = 'DELETE' THEN to_jsonb(OLD) ELSE to_jsonb(NEW) END;
    v_record_id := v_row ->> 'id';
    v_sid := (CASE WHEN TG_TABLE_NAME = 'subject'
                   THEN v_row ->> 'id'
                   ELSE v_row ->> TG_ARGV[0]
              END)::uuid;

    IF (TG_OP = 'INSERT') THEN
        v_step := 'INSERT: normalize';
        new_json := v_row;
        IF new_json ? 'categories' THEN
            new_json := jsonb_set(new_json, '{categories}',
                COALESCE(normalize_categories(new_json -> 'categories'), 'null'::jsonb));
        END IF;

        v_step := 'INSERT: insert into subject_history';
        INSERT INTO subject_history (subject_id, table_name, record_id, op, old_diff, new_diff)
        VALUES (v_sid, TG_TABLE_NAME, v_record_id, 'I', NULL, jsonb_strip_nulls(new_json));
        RETURN NEW;

    ELSIF (TG_OP = 'UPDATE') THEN
        old_json := to_jsonb(OLD);
        new_json := to_jsonb(NEW);

        FOR k IN SELECT jsonb_object_keys(new_json)
        LOOP
            IF k IN ('updated_at', 'dt_upd') THEN
                CONTINUE;
            END IF;

            v_step := 'UPDATE: OLD, key=' || k;
            BEGIN
                val_text := old_json ->> k; 

                 
                IF k = 'categories' THEN
                    v_old := normalize_categories(old_json -> k);
                ELSIF val_text IS NOT NULL AND (val_text LIKE '{%' OR val_text LIKE '[%') THEN
                    v_old := val_text::jsonb;
                ELSE
                    v_old := old_json -> k;
                END IF;
            EXCEPTION WHEN others THEN
                v_old := old_json -> k;
            END;

            v_step := 'UPDATE: NEW, key=' || k;
            BEGIN
                val_text := new_json ->> k;
                IF k = 'categories' THEN
                    v_new := normalize_categories(new_json -> k);
                ELSIF val_text IS NOT NULL AND (val_text LIKE '{%' OR val_text LIKE '[%') THEN
                    v_new := val_text::jsonb;
                ELSE
                    v_new := new_json -> k;
                END IF;
            EXCEPTION WHEN others THEN
                v_new := new_json -> k;
            END;

            IF v_old IS DISTINCT FROM v_new THEN
                diff_old := diff_old || jsonb_build_object(k, v_old);
                diff_new := diff_new || jsonb_build_object(k, v_new);
            END IF;
        END LOOP;

        v_step := 'UPDATE: insert into subject_history';
        IF diff_new <> '{}'::JSONB THEN
            INSERT INTO subject_history (subject_id, table_name, record_id, op, old_diff, new_diff)
            VALUES (v_sid, TG_TABLE_NAME, v_record_id, 'U', diff_old, diff_new);
        END IF;
        RETURN NEW;

    ELSIF (TG_OP = 'DELETE') THEN
        v_step := 'DELETE: normalize';
        old_json := v_row;
        IF old_json ? 'categories' THEN
            old_json := jsonb_set(old_json, '{categories}',
                COALESCE(normalize_categories(old_json -> 'categories'), 'null'::jsonb));
        END IF;

        v_step := 'DELETE: insert into subject_history';
        INSERT INTO subject_history (subject_id, table_name, record_id, op, old_diff, new_diff)
        VALUES (v_sid, TG_TABLE_NAME, v_record_id, 'D', jsonb_strip_nulls(old_json), NULL);
        RETURN OLD;
    END IF;

    RETURN NULL;

EXCEPTION WHEN others THEN
    GET STACKED DIAGNOSTICS
        v_err_state   = RETURNED_SQLSTATE,
        v_err_msg     = MESSAGE_TEXT,
        v_err_detail  = PG_EXCEPTION_DETAIL,
        v_err_context = PG_EXCEPTION_CONTEXT;

    INSERT INTO subject_history_log
        (subject_id, tg_op, step, sqlstate, message, detail, context)
    VALUES
        (v_sid::text, TG_TABLE_NAME || ' ' || TG_OP, v_step,
         v_err_state, v_err_msg, v_err_detail, v_err_context);

    IF TG_OP = 'DELETE' THEN
        RETURN OLD;
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;



CREATE TRIGGER trg_phone_numbers_history
AFTER INSERT OR UPDATE OR DELETE ON phone_numbers
FOR EACH ROW EXECUTE FUNCTION log_subject_diff('subject_id');

CREATE TRIGGER trg_cards_history
AFTER INSERT OR UPDATE OR DELETE ON cards
FOR EACH ROW EXECUTE FUNCTION log_subject_diff('subject_id');

CREATE TRIGGER trg_ibans_new_history
AFTER INSERT OR UPDATE OR DELETE ON ibans_new
FOR EACH ROW EXECUTE FUNCTION log_subject_diff('subject_id');

CREATE TRIGGER trg_incidents_subject_history
AFTER INSERT OR UPDATE OR DELETE ON incidents_subject
FOR EACH ROW EXECUTE FUNCTION log_subject_diff('subject_id');



ALTER TABLE subject_history ADD COLUMN IF NOT EXISTS table_name TEXT;
ALTER TABLE subject_history ADD COLUMN IF NOT EXISTS record_id TEXT;

UPDATE subject_history SET table_name = 'subject' WHERE table_name IS NULL;
