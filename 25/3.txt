CREATE OR REPLACE FUNCTION public.report_categories(
  d_start timestamp with time zone,
  d_end timestamp with time zone
)
RETURNS TABLE (
  n integer,
  category text,
  fl_start bigint,
  fl_arrived bigint,
  fl_changed bigint,
  fl_removed bigint,
  fl_end bigint,
  fl_now bigint,
  ul_start bigint,
  ul_arrived bigint,
  ul_changed bigint,
  ul_removed bigint,
  ul_end bigint,
  ul_now bigint
)
LANGUAGE sql
STABLE
AS $function$
WITH cats(n, category) AS (
  VALUES
    (1, 'WHITE_LIST'),
    (2, 'POTENTIAL_VICTIM'),
    (3, 'GREY'),
    (4, 'CASINO_CEPI_GREY'),
    (5, 'CASINO_GREY'),
    (6, 'DRUG_GREY'),
    (7, 'PYRAMID_GREY'),
    (8, 'WITNESS'),
    (9, 'VICTIM'),
    (10, 'DROPPER'),
    (11, 'PYRAMID_CEPI_GREY'),
    (12, 'BLACK'),
    (13, 'DRUG_CEPI_GREY')
),
base AS (
  SELECT *
  FROM subject_category_item
  WHERE created < d_end
),
new_ps AS (
  SELECT subject_id, array_agg(DISTINCT category) AS new_cats
  FROM base
  WHERE created >= d_start
  GROUP BY subject_id
),
f AS (
  SELECT
    b.subject_type,
    b.category,
    b.subject_id,
    (b.created < d_start AND (b.deleted IS NULL OR b.deleted >= d_start)) AS on_start,
    (b.created >= d_start) AS arrived,
    (b.deleted >= d_start AND b.deleted < d_end) AS del_in,
    COALESCE(cardinality(array_remove(n.new_cats, b.category)) > 0, false) AS has_new,
    (b.deleted IS NULL OR b.deleted >= d_end) AS on_end,
    NOT b.is_deleted AS now_
  FROM base b
  LEFT JOIN new_ps n USING (subject_id)
)
SELECT
  c.n,
  c.category,
  count(f.subject_id) FILTER (WHERE f.subject_type = 'PERSON' AND f.on_start),
  count(f.subject_id) FILTER (WHERE f.subject_type = 'PERSON' AND f.arrived),
  count(f.subject_id) FILTER (WHERE f.subject_type = 'PERSON' AND f.del_in AND f.has_new),
  count(f.subject_id) FILTER (WHERE f.subject_type = 'PERSON' AND f.del_in AND NOT f.has_new),
  count(f.subject_id) FILTER (WHERE f.subject_type = 'PERSON' AND f.on_end),
  count(f.subject_id) FILTER (WHERE f.subject_type = 'PERSON' AND f.now_),
  count(f.subject_id) FILTER (WHERE f.subject_type = 'ORGANIZATION' AND f.on_start),
  count(f.subject_id) FILTER (WHERE f.subject_type = 'ORGANIZATION' AND f.arrived),
  count(f.subject_id) FILTER (WHERE f.subject_type = 'ORGANIZATION' AND f.del_in AND f.has_new),
  count(f.subject_id) FILTER (WHERE f.subject_type = 'ORGANIZATION' AND f.del_in AND NOT f.has_new),
  count(f.subject_id) FILTER (WHERE f.subject_type = 'ORGANIZATION' AND f.on_end),
  count(f.subject_id) FILTER (WHERE f.subject_type = 'ORGANIZATION' AND f.now_)
FROM cats c
LEFT JOIN f ON f.category = c.category
GROUP BY c.n, c.category
ORDER BY c.n;
$function$;
