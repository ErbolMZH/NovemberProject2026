package repository

import (
	"context"
	"your_project/models"
	"github.com/jmoiron/sqlx"
)

type ReportRepository struct {
	db *sqlx.DB
}

func NewReportRepository(db *sqlx.DB) *ReportRepository {
	return &ReportRepository{db: db}
}

func (r *ReportRepository) GetCategoriesReport(ctx context.Context, startDate, endDate string) ([]models.CategoryReportRow, error) {
	// Вызываем вашу функцию из PostgreSQL. 
	// Проверьте имя первого столбца в определении функции (вместо "id" может быть "ord" или "num")
	query := `
		SELECT 
			*
		FROM report_categories($1::timestamptz, $2::timestamptz)
	`
	
	rows, err := r.db.QueryxContext(ctx, query, startDate, endDate)
	if err != nil {
		return nil, err
	}
	defer rows.Close()

	var result []models.CategoryReportRow
	for rows.Next() {
		var row models.CategoryReportRow
		// Если первый столбец в функции не имеет алиаса "id", используйте rows.Scan напрямую:
		err := rows.Scan(
			&row.ID,
			&row.Category,
			&row.FlStart, &row.FlArrived, &row.FlChanged, &row.FlRemoved, &row.FlEnd, &row.FlNow,
			&row.UlStart, &row.UlArrived, &row.UlChanged, &row.UlRemoved, &row.UlEnd, &row.UlNow,
		)
		if err != nil {
			return nil, err
		}
		result = append(result, row)
	}

	return result, rows.Err()
}

package handler

import (
	"net/http"
	"time"
	"your_project/repository"
	"github.com/labstack/echo/v4"
)

type ReportHandler struct {
	repo *repository.ReportRepository
}

func NewReportHandler(repo *repository.ReportRepository) *ReportHandler {
	return &ReportHandler{repo: repo}
}

func (h *ReportHandler) GetCategoriesReport(c echo.Context) error {
	startDate := c.QueryParam("start_date")
	endDate := c.QueryParam("end_date")

	if startDate == "" || endDate == "" {
		return c.JSON(http.StatusBadRequest, map[string]string{
			"error": "Параметры start_date и end_date обязательны",
		})
	}

	// Валидация формата времени (поддержка ISO-строк с фронтенда)
	if _, err := time.Parse(time.RFC3339, startDate); err != nil {
		return c.JSON(http.StatusBadRequest, map[string]string{
			"error": "Неверный формат start_date (ожидается RFC3339)",
		})
	}
	if _, err := time.Parse(time.RFC3339, endDate); err != nil {
		return c.JSON(http.StatusBadRequest, map[string]string{
			"error": "Неверный формат end_date (ожидается RFC3339)",
		})
	}

	data, err := h.repo.GetCategoriesReport(c.Request().Context(), startDate, endDate)
	if err != nil {
		return c.JSON(http.StatusInternalServerError, map[string]string{
			"error": "Ошибка при формировании отчета: " + err.Error(),
		})
	}

	return c.JSON(http.StatusOK, data)
}

    computed: {
        totals() {
            const fields = [
                'fl_start', 'fl_arrived', 'fl_changed', 'fl_removed', 'fl_end', 'fl_now',
                'ul_start', 'ul_arrived', 'ul_changed', 'ul_removed', 'ul_end', 'ul_now'

            const sum = {};

    methods: {
        formatDatetimeLocal(date) {
            const pad = (n) => String(n).padStart(2, '0');
            return `${date.getFullYear()}-${pad(date.getMonth() + 1)}-${pad(date.getDate())}T${pad(date.getHours())}:${pad(date.getMinutes())}:${pad(date.getSeconds())}`;

        toPgTimestamp(localStr) {
            if (!localStr) return '';
            // Преобразуем "2026-08-27T18:53:00" в "2026-08-27 18:53:00+05:00"
            const normalized = localStr.length === 16 ? `${localStr}:00` : localStr;
            return `${normalized.replace('T', ' ')}+05:00`;

        async fetchReport() {

                        start_date: this.toPgTimestamp(this.startDate),
                        end_date: this.toPgTimestamp(this.endDate)
                    }
                });

                console.error("Ошибка при загрузке отчёта:", error);
                alert("Ошибка загрузки отчёта");
            } finally {
                this.isLoading = false;
            }

            const startOfDay = new Date(now.getFullYear(), now.getMonth(), now.getDate(), 0, 0, 0);
            this.startDate = this.formatDatetimeLocal(startOfDay);
            this.endDate = this.formatDatetimeLocal(now);
            this.fetchReport();
        }
