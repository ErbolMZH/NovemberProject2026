CREATE OR REPLACE FUNCTION log_subject_diff()
RETURNS TRIGGER AS $$
DECLARE
    old_json JSONB;
    new_json JSONB;
    diff_old JSONB := '{}'::JSONB;
    diff_new JSONB := '{}'::JSONB;
    k TEXT;
    v_old JSONB;
    v_new JSONB;
    val_text TEXT;

    v_step        TEXT := 'start';
    v_subject_id  TEXT;
    v_err_state   TEXT;
    v_err_msg     TEXT;
    v_err_detail  TEXT;
    v_err_context TEXT;
BEGIN
    IF (TG_OP = 'INSERT') THEN
        v_subject_id := NEW.id::text;

        v_step := 'INSERT: to_jsonb(NEW)';
        new_json := to_jsonb(NEW);

        v_step := 'INSERT: normalize_categories';
        IF new_json ? 'categories' THEN
            new_json := jsonb_set(new_json, '{categories}',
                COALESCE(normalize_categories(new_json -> 'categories'), 'null'::jsonb));
        END IF;

        v_step := 'INSERT: insert into subject_history';
        INSERT INTO subject_history (subject_id, op, old_diff, new_diff)
        VALUES (NEW.id, 'I', NULL, jsonb_strip_nulls(new_json));
        RETURN NEW;

    ELSIF (TG_OP = 'UPDATE') THEN
        v_subject_id := NEW.id::text;

        v_step := 'UPDATE: to_jsonb(OLD/NEW)';
        old_json := to_jsonb(OLD);
        new_json := to_jsonb(NEW);

        FOR k IN SELECT jsonb_object_keys(new_json)
        LOOP
            IF k IN ('updated_at', 'dt_upd') THEN
                CONTINUE;
            END IF;

            -- OLD
            v_step := 'UPDATE: OLD, key=' || k;
            BEGIN
                val_text := old_json ->> k;
                IF k = 'categories' THEN
                    v_old := normalize_categories(old_json -> k);
                ELSIF val_text IS NOT NULL AND (val_text LIKE '{%' OR val_text LIKE '[%') THEN
                    v_old := val_text::jsonb;
                ELSE
                    v_old := old_json -> k;
                END IF;
            EXCEPTION WHEN others THEN
                GET STACKED DIAGNOSTICS
                    v_err_state = RETURNED_SQLSTATE,
                    v_err_msg   = MESSAGE_TEXT;
                INSERT INTO subject_history_log (subject_id, tg_op, step, sqlstate, message)
                VALUES (v_subject_id, TG_OP, v_step || ' (ошибка парсинга, взято как есть)',
                        v_err_state, v_err_msg);
                v_old := old_json -> k;
            END;

            -- NEW
            v_step := 'UPDATE: NEW, key=' || k;
            BEGIN
                val_text := new_json ->> k;
                IF k = 'categories' THEN
                    v_new := normalize_categories(new_json -> k);
                ELSIF val_text IS NOT NULL AND (val_text LIKE '{%' OR val_text LIKE '[%') THEN
                    v_new := val_text::jsonb;
                ELSE
                    v_new := new_json -> k;
                END IF;
            EXCEPTION WHEN others THEN
                GET STACKED DIAGNOSTICS
                    v_err_state = RETURNED_SQLSTATE,
                    v_err_msg   = MESSAGE_TEXT;
                INSERT INTO subject_history_log (subject_id, tg_op, step, sqlstate, message)
                VALUES (v_subject_id, TG_OP, v_step || ' (ошибка парсинга, взято как есть)',
                        v_err_state, v_err_msg);
                v_new := new_json -> k;
            END;

            v_step := 'UPDATE: compare, key=' || k;
            IF v_old IS DISTINCT FROM v_new THEN
                diff_old := diff_old || jsonb_build_object(k, v_old);
                diff_new := diff_new || jsonb_build_object(k, v_new);
            END IF;
        END LOOP;

        v_step := 'UPDATE: insert into subject_history';
        IF diff_new <> '{}'::JSONB THEN
            INSERT INTO subject_history (subject_id, op, old_diff, new_diff)
            VALUES (NEW.id, 'U', diff_old, diff_new);
        END IF;
        RETURN NEW;

    ELSIF (TG_OP = 'DELETE') THEN
        v_subject_id := OLD.id::text;

        v_step := 'DELETE: to_jsonb(OLD)';
        old_json := to_jsonb(OLD);

        v_step := 'DELETE: normalize_categories';
        IF old_json ? 'categories' THEN
            old_json := jsonb_set(old_json, '{categories}',
                COALESCE(normalize_categories(old_json -> 'categories'), 'null'::jsonb));
        END IF;

        v_step := 'DELETE: insert into subject_history';
        INSERT INTO subject_history (subject_id, op, old_diff, new_diff)
        VALUES (OLD.id, 'D', jsonb_strip_nulls(old_json), NULL);
        RETURN OLD;
    END IF;

    RETURN NULL;

EXCEPTION WHEN others THEN
    GET STACKED DIAGNOSTICS
        v_err_state   = RETURNED_SQLSTATE,
        v_err_msg     = MESSAGE_TEXT,
        v_err_detail  = PG_EXCEPTION_DETAIL,
        v_err_context = PG_EXCEPTION_CONTEXT;

    INSERT INTO subject_history_log
        (subject_id, tg_op, step, sqlstate, message, detail, context)
    VALUES
        (v_subject_id, TG_OP, v_step, v_err_state, v_err_msg, v_err_detail, v_err_context);

    -- ошибка истории не должна ломать основную операцию над subject
    IF TG_OP = 'DELETE' THEN
        RETURN OLD;
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;
