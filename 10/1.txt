// Модель для категории
type AFCategoryModel struct {
    Code string `json:"code" db:"code"`
    Name string `json:"name" db:"name"`
}

// Запрос в БД
func GetCategoriesDictionary() ([]AFCategoryModel, error) {
    var categories []AFCategoryModel
    query := `select code, name from category_dictionary order by name asc`
    err := pg.DB().Select(&categories, query)
    if err != nil {
        return nil, err
    }
    return categories, nil
}

// Echo Handler
func GetCategoriesHandler(c echo.Context) error {
    categories, err := GetCategoriesDictionary()
    if err != nil {
        log.Printf("Ошибка при получении категорий: %v", err)
        return c.JSON(http.StatusInternalServerError, map[string]interface{}{
            "success": false,
            "error":   err.Error(),
        })
    }
    return c.JSON(http.StatusOK, map[string]interface{}{
        "success": true,
        "data":    categories,
    })
}


async fetchCategories() {
    try {
        const resp = await axios.get('/incidents/categories');
        this.categories = resp.data.data || [];
    } catch (error) {
        console.error("Ошибка при загрузке категорий:", error);
    }
},
