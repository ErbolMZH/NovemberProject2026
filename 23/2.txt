WITH params AS (
    SELECT
        '2026-01-01 00:00:00+00'::timestamptz AS d_from,
        '2026-02-01 00:00:00+00'::timestamptz AS d_to
),
parsed_raw AS (
    SELECT
        s.id,
        s."subjectType" AS subj_type, -- проверьте точное название/значения типа субъекта в вашей БД[span_3](start_span)[span_3](end_span)[span_4](start_span)[span_4](end_span)
        elem,
        CASE
            WHEN left(btrim(elem), 1) = '{' THEN elem::jsonb
            ELSE NULL::jsonb
        END AS j
    FROM subject s -- проверьте имя схемы/таблицы[span_5](start_span)[span_5](end_span)[span_6](start_span)[span_6](end_span)
    CROSS JOIN LATERAL unnest(s.categories) AS elem -- проверьте имя колонки массива[span_7](start_span)[span_7](end_span)[span_8](start_span)[span_8](end_span)
    WHERE elem IS NOT NULL
      AND btrim(elem) <> ''
),
parsed AS (
    SELECT
        id,
        subj_type,
        CASE
            WHEN j IS NOT NULL THEN j->>'category'
            ELSE btrim(elem)
        END AS category,
        CASE
            WHEN j IS NOT NULL THEN COALESCE((j->>'isDeleted')::boolean, false)
            ELSE false
        END AS is_deleted,
        CASE
            WHEN j IS NOT NULL AND NULLIF(j->>'createdDate', '') IS NOT NULL
                THEN (j->>'createdDate')::timestamptz
        END AS created_dt,
        CASE
            WHEN j IS NOT NULL AND NULLIF(j->>'deletedDate', '') IS NOT NULL
                THEN (j->>'deletedDate')::timestamptz
        END AS deleted_dt
    FROM parsed_raw
)
SELECT
    p.category,
    COUNT(*) FILTER (
        WHERE p.created_dt < pr.d_from
          AND (p.deleted_dt IS NULL OR p.deleted_dt >= pr.d_from)
    ) AS na_nachalo_perioda,
    COUNT(*) FILTER (
        WHERE p.created_dt >= pr.d_from
          AND p.created_dt < pr.d_to
    ) AS postupilo_za_period,
    COUNT(*) FILTER (
        WHERE p.deleted_dt >= pr.d_from
          AND p.deleted_dt < pr.d_to
    ) AS udalena_za_period,
    COUNT(*) FILTER (
        WHERE p.created_dt < pr.d_to
          AND (p.deleted_dt IS NULL OR p.deleted_dt >= pr.d_to)
    ) AS na_konec_perioda,
    COUNT(*) FILTER (
        WHERE NOT p.is_deleted
          AND p.deleted_dt IS NULL
    ) AS seychas
FROM parsed p
CROSS JOIN params pr
WHERE p.category IS NOT NULL
  AND p.category <> ''
GROUP BY p.category
ORDER BY p.category;
